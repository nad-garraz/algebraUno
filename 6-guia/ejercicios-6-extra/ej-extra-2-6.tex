\begin{enunciado}{\ejExtra}
  Sea $w \en G_{14}.$
  Hallar todos los posibles valores de
  $w^7 + \sumatoria{j=7}{140}w^{2j}$
\end{enunciado}

Voy a usar que:
$
  \llave{l}{
    w \en G_n \entonces \sumatoria{k=0}{n-1} w^k = 0 \\
    \text{Si } m \divideA n \entonces G_m \subseteq G_n
  }
$

\textit{Si $w = 1$}:

$$
  \ub{w^7}{= \green{1}} + \sumatoria{j=7}{140} \ub{w^{2j}}{=1} =
  \green{1} + (\ub{1+1+\dots+1}{ = 134}) =
  1 + 134 =
  135
$$

\textit{Si $w = -1$}:

$$
  \ub{w^7}{= \green{-1}} + \sumatoria{j=7}{140} \ub{(w^j)^2}{= 1} =
  \green{-1} + (\ub{1+1+\dots+1}{ = 134}) =
  -1 + 134 =
  133
$$

\textit{Si $w \distinto \pm 1$}:

Como $w \en G_{14}$ es de la forma:
$$
  w = e^{i\frac{2k\pi}{14}} \text{ con } k \en \enteros_{[0,13]}.
$$
y se desprende inmediatamente que $w^{14} \igual{$\llamada1$} 1$
Entonces es fácil ver que:
$$
  \textstyle
  w^2 =
  (
  e^{i\frac{2k\pi}{14}}
  )^2 =
  \ub{
    e^{i\frac{2\pi}{7}\cdot k}
  }{
    \en G_7
  }
  \Entonces{Por lo tanto}
  \sumatoria{j=0}{6} (w^2)^j \igual{$\llamada2$} 0
$$
Ahora con la cuenta del enunciado:
$$
  \begin{array}{rcl}
    w^7 + \sumatoria{j=7}{140} w^{2j}
     & \igual{\red{!!}}    &
    w^7 + \sumatoria{j=0}{140} (w^2)^j -
    \ob{
      \sumatoria{j=0}{6} (w^2)^j
    }{
      \igual{$\llamada2$} 0
    }                                     \\
     & \igual{\red{!}}     &
    w^7 + \frac{(w^2)^{141} - 1}{w^2 - 1} \\
     & \igual{\red{!}}     &
    w^7 + \frac{
      (w^{14})^{10}\cdot w^2 - 1
    }{w^2 - 1}                            \\
     & \igual{$\llamada1$} &
    \cajaResultado{
      w^7 + 1
    }\llamada3
  \end{array}
$$
Los posibles valores que puede tomar la expresión dependen de $w^7$ con $w \en G_{14}$:
$$
  \begin{tikzpicture}[baseline=0,scale = 2.3, every node/.style={font=\tiny}]
    \draw[ultra thin,->,gray] (-1.5,0) -- (1.7,0) node[below] {Re};

          \draw[OliveGreen] (\x*360/14:1) -- ({(\x+1)*360/14}:1);

        \fi
      }
  \end{tikzpicture}
  \qquad
  \begin{tikzpicture}[baseline=0,scale = 2.3, every node/.style={font=\tiny}]

          \fi
        \fi

          \draw[cyan] ({\x*360/7+360/14}:1) -- ({(\x+1)*360/7+360/14}:1);

  \end{tikzpicture}
$$
De los gráficos observo que:
$$
  \begin{array}{rccl}
    \text{si} & w \en G_7          & \entonces & w^7 = 1  \\
    \text{si} & w \en G_{14} - G_7 & \entonces & w^7 = -1
  \end{array}
$$
Por lo tanto en la expresión $\llamada3$ puedo obtener dependiendo del $w$:
$$
  \llave{ccl}{
    w \en G_7          & \to & \magenta{1} + 1 = 2  \\
    w \en G_{14} - G_7 & \to & \magenta{-1} + 1 = 0
  }
$$
Finalmente los posibles valores serán:
$$
  \cajaResultado{
    \set{0,2,133, 135}
  }
$$
